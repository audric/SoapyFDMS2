cmake_minimum_required(VERSION 3.16)
project(SoapyFdms2 CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
find_package(SoapySDR CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
pkg_check_modules(LIBUSB REQUIRED IMPORTED_TARGET libusb-1.0)
SOAPY_SDR_MODULE_UTIL(TARGET fdms2Support SOURCES SoapyFdms2.cpp LIBRARIES PkgConfig::LIBUSB Threads::Threads)

option(FDMS2_BUILD_TESTS "Build the protocol test" ON)
if(FDMS2_BUILD_TESTS)
    add_executable(fdms2_protocol_test fdms2_protocol_test.cpp)
    target_compile_options(fdms2_protocol_test PRIVATE -Wall -Wextra)
endif()
